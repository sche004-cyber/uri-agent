import 'package:flutter/material.dart';

class ProviderEntry {
  final String providerId;
  final String displayName;
  final String adapter;
  final String baseUrl;
  final bool configured;
  final String? lastFour;
  final bool available;

  ProviderEntry({
    required this.providerId,
    required this.displayName,
    required this.adapter,
    required this.baseUrl,
    required this.configured,
    this.lastFour,
    required this.available,
  });

  factory ProviderEntry.fromJson(Map<String, dynamic> json) {
    return ProviderEntry(
      providerId: json['provider_id'] ?? '',
      displayName: json['display_name'] ?? '',
      adapter: json['adapter'] ?? '',
      baseUrl: json['base_url'] ?? '',
      configured: json['configured'] ?? false,
      lastFour: json['last_four'] as String?,
      available: json['available'] ?? true,
    );
  }
}

class ProviderKeyResult {
  final String providerId;
  final bool configured;
  final String lastFour;

  ProviderKeyResult({
    required this.providerId,
    required this.configured,
    required this.lastFour,
  });

  factory ProviderKeyResult.fromJson(Map<String, dynamic> json) {
    return ProviderKeyResult(
      providerId: json['provider_id'] ?? '',
      configured: json['configured'] ?? false,
      lastFour: json['last_four'] ?? '',
    );
  }
}

abstract class UriClient {
  Future<List<ProviderEntry>> listProviders();
  Future<ProviderKeyResult?> submitProviderKey(String providerId, String apiKey);
  Future<bool> updateProviderConfig(String providerId, {String? baseUrl, String? model});
}

import 'dart:convert';
import 'package:http_client/http_client.dart'; // Assuming a standard HTTP client or http package
import 'uri_client.dart';

class HttpUriClient implements UriClient {
  final String baseUrl;
  HttpUriClient(this.baseUrl);

  static const _jsonHeaders = {'Content-Type': 'application/json', 'Accept': 'application/json'};

  @override
  Future<List<ProviderEntry>> listProviders() async {
    final response = await _get('$baseUrl/providers');
    if (response.statusCode == 200) {
      final data = jsonDecode(response.body);
      final List decoded = data['providers'] ?? [];
      return decoded.map((e) => ProviderEntry.fromJson(e)).toList();
    }
    throw Exception('Failed to fetch providers');
  }

  @override
  Future<ProviderKeyResult?> submitProviderKey(String providerId, String apiKey) async {
    final response = await _post('$baseUrl/providers/keys', {
      'provider_id': providerId,
      'api_key': apiKey,
    });

    if (response.statusCode == 200 || response.statusCode == 201) {
      final data = jsonDecode(response.body);
      return ProviderKeyResult.fromJson(data);
    }
    return null;
  }

  @override
  Future<bool> updateProviderConfig(String providerId, {String? baseUrl, String? model}) async {
    final response = await _put('$baseUrl/providers/config', {
      'provider_id': providerId,
      'base_url': baseUrl,
      'model': model,
    });
    return response.statusCode == 200;
  }

  // Helper methods abstracted for the example
  Future<dynamic> _get(String url) async { /* Implementation of GET */ }
  Future<dynamic> _post(String url, Map body) async { /* Implementation of POST */ }
  Future<dynamic> _put(String url, Map body) async { /* Implementation of PUT */ }
}

// uri_ui/lib/screens/settings/providers_screen.dart
import 'package:flutter/material.dart';
import '../../services/app_state.dart';
import '../../services/uri_client.dart';

class ProvidersScreen extends StatefulWidget {
  final AppState state;
  const ProvidersScreen({super.key, required this.state});

  @override
  State<ProvidersScreen> createState() => _ProvidersScreenState();
}

class _ProvidersScreenState extends State<ProvidersScreen> {
  List<ProviderEntry> _providers = [];
  bool _isLoading = true;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    _refreshProviders();
  }

  void _refreshProviders() async {
    setState(() => _isLoading = true);
    try {
      final list = await widget.state.listProviders();
      setState(() {
        _providers = list;
        _isLoading = false;
      });
    } catch (e) {
      setState(() {
        _errorMessage = "Failed to load providers.";
        _isLoading = false;
      });
    }
  }

  void _handleSubmitKey(String providerId, String apiKey) async {
    final result = await widget.state.submitProviderKey(providerId, apiKey);
    if (result != null) {
      setState(() => _errorMessage = null);
      _refreshProviders(); // Refresh list to show updated status/last 4
    } else {
      setState(() => _errorMessage = "Failed to submit API key. Please try again.");
    }
  }

  @override
  Widget build(BuildContext context) {
    if (_isLoading) return const Center(child: CircularProgressIndicator());
    if (_errorMessage != null) return Center(child: Text(_errorMessage!, style: const TextStyle(color: Colors.red)));

    return Scaffold(
      app_bar: AppBar(title: const Text("Model Providers")),
      body: ListView.builder(
        itemCount: _providers.length,
        itemBuilder: (context, index) {
          final item = _providers[index];
          return ListTile(
            title: Text(item.displayName),
            subtitle: Text("${item.adapter} • ${item.baseUrl}"),
            trailing: Row(
              mainAxisSize: MainAxisSize.min,
              children: [
                if (item.configured) 
                  _buildStatusPill("Configured", Colors.green),
                const SizedBox(width: 8),
                if (item.configured) Text("...${item.lastFour}"),
                const SizedBox(width: 12),
                Icon(item.available ? Icons.check_circle : Icons.warning, 
                     color: item.available ? Colors.green : Colors.orange),
              ],
            ),
            onTap: () => _showConfigDialog(item),
          );
        },
      ),
    );
  }

  Widget _buildStatusPill(String text, Color color) {
    return Container(
      padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 2),
      decoration: BoxDecoration(color: color, borderRadius: BorderRadius.circular(10)),
      child: Text(text, style: const TextStyle(color: Colors.white, fontSize: 12)),
    );
  }

  void _showConfigDialog(ProviderEntry item) {
    final keyController = TextEditingController();
    showDialog(
      context: context,
      builder: (context) => AlertDialog(
        title: Text("Configure ${item.displayName}"),
        content: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            TextField(
              controller: keyController,
              obscureText: true,
              decoration: const InputDecoration(labelText: "Enter API Key"),
            ),
            const SizedBox(height: 20),
            ElevatedButton(
              onPressed: () async {
                await _handleSubmitKey(item.providerId, keyController.text);
                Navigator.pop(context);
              },
              child: const Text("Confirm Key"),
            )
          ],
        ),
      ),
    );
  }
}
